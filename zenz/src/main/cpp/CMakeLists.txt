cmake_minimum_required(VERSION 3.22.1)
project(zenz_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../.tools/zenz-source/llama.cpp-88b97a47dc7f5892e2d5a6856fbe9cfe237f9e5c")
if(NOT EXISTS "${LLAMA_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run scripts/prepare-zenz.ps1 before building")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(zenz SHARED zenz_bridge.cpp)
target_include_directories(zenz PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include")
if(MSVC)
    target_compile_options(zenz PRIVATE /O2 /utf-8 "/FI${CMAKE_CURRENT_LIST_DIR}/../../test/native/msvc-compat.h")
else()
    target_compile_options(zenz PRIVATE -O3)
endif()
target_link_libraries(zenz PRIVATE llama ggml)
if(ANDROID)
    target_link_libraries(zenz PRIVATE log)
else()
    find_package(JNI REQUIRED)
    target_include_directories(zenz PRIVATE ${JNI_INCLUDE_DIRS})
endif()
